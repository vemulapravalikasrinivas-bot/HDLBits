Although Lemmings can walk, fall, and dig, Lemmings aren't invulnerable. If a Lemming falls for too long then hits the ground, it can splatter. In particular, if a Lemming falls for more than 20 clock cycles then hits the ground, it will splatter and cease walking, falling, or digging (all 4 outputs become 0), forever (Or until the FSM gets reset). There is no upper limit on how far a Lemming can fall before hitting the ground. Lemmings only splatter when hitting the ground; they do not splatter in mid-air.

Extend your finite state machine to model this behaviour.


module top_module(
    input clk,
    input areset,
    input bump_left,
    input bump_right,
    input ground,
    input dig,
    output walk_left,
    output walk_right,
    output aaah,
    output digging
);

    
parameter L = 0, R=1, FL=2,FR=3,DL=4,DR=5,SP=6;
    reg [2:0] state, nextstate;
always @(posedge clk or posedge areset)
    if (areset) state <= L;
    else state <= nextstate;
    reg [4:0] count;
always @(posedge clk or posedge areset)
    if (areset) count <= 0;
    else if (state == FL ||state == FR) count <= count < 20 ? count + 1 : count;
    else if (state == L || state == R) count <= 0;
    else count <= count;
always @(*)
    case (state)
          L: nextstate = ~ground ? FL : (dig ? DL : (bump_left ? R : L));
        R: nextstate = ~ground ? FR : (dig ? DR : (bump_right ? L : R));
        FL: nextstate = ~ground ? FL : (count >= 20 ? SP : L);
        FR:  nextstate = ~ground ? FR : (count >= 20 ? SP : R);
        DL:  nextstate = ground ? DL : FL;
        DR:  nextstate = ground ? DR : FR;
        SP: nextstate = SP;
      
            default:         nextstate = L;
    endcase
    assign walk_left = state == SP ? 1'b0 : (state == L);
    assign walk_right = state == SP ? 1'b0 : (state == R);
    assign aaah = state == SP ? 1'b0 : (state == FL || state == FR);
    assign digging = state== SP ? 1'b0 : (state == DL || state == DR);

endmodule
