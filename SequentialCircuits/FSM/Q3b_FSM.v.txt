Given the state-assigned table shown below, implement the finite-state machine. Reset should reset the FSM to state 000.

Present state
y[2:0]	Next state Y[2:0]	Output z
	x=0	x=1
000	000	001		0
001	001	100		0
010	010	001		0
011	001	010		1
100	011	100		1



module top_module (
    input clk,
    input reset,   // Synchronous reset
    input x,
    output z
);
    parameter s0=0, s1=1, s2=2, s3=3, s4=4;
    reg [2:0] state,ns;
    
    always @(posedge clk) begin
        if(reset) 
            state<=s0;
        else
            state<=ns;
    end
    
    always @(*) begin
        case(state) 
            s0: ns=x?s1:s0;
            s1: ns=x?s4:s1;
            s2: ns=x?s1:s2;
            s3: ns=x?s2:s1;
            s4: ns=x?s4:s3;
            default:ns=s0;
        endcase
    end
    assign z=(state==s4)||(state==s3);    

endmodule
